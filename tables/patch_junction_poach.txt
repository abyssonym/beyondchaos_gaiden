.def    poach-rare-rate                 20

0213a1: .jml $main
:       .nop .nop

$main
:       .php
:       .cpy 08
:       .bcc exit
:       .lda.dp b5
:       .and 00ff
:       .bne exit

# here we get the character index from the stack, then store it in $EE
# which would have happened anyway, we're just doing it a little sooner
:       .sep 20
:       .lda.sr 05
:       .sta.dp ee
:       .rep 30

:       .and 00ff
:       .cmp 0008
:       .bcs exit

:       .phx
:       .tax

# check for imp status
:       .lda.addr.y 3ee4
:       .bit 0020
:       .beq check-poach

:       .lda {{jun-index-imp-harvest}} 00
:       .jsl $jun-checker
:       .bne imp-harvest

.label check-poach
:       .lda {{jun-index-poach}} 00
:       .jsl $jun-checker
:       .bne poach

.label pull-and-exit
:       .plx
.label exit
:       .plp
:       .lda 0080
:       .jml $return

.label imp-harvest
:       .jsr.addr $get-steal-or-drop,2
# always choose rare item
:       .jsr.addr $obtain-item,2
:       .bra pull-and-exit

.label poach
:       .jsr.addr $get-steal-or-drop,2
# choose rare item 1/8 of the time
:       .tax
:       .jsl $jun-rng2
:       .and 00ff
:       .cmp poach-rare-rate 00
:       .bcc poach-choose-rare
:       .txa
:       .xba
:       .bra poach-item-chosen
.label poach-choose-rare
:       .txa
.label poach-item-chosen
:       .jsr.addr $obtain-item,2
:       .bra pull-and-exit

$get-steal-or-drop
:       .lda.addr.y $monster-indexes,2
:       .asl .asl
:       .tax
:       .jsl $jun-rng2
:       .lsr
:       .bcs choose-drop
:       .lda.long.x $monster-steals
:       .rts
.label choose-drop
:       .lda.long.x $monster-drops
:       .rts
