.addr   main                            628200
.addr   reentry-address                 023292

.addr   pointer-address                 {{jun-global-command-fixed-damage}}

$main
:       .tyx
:       .sep 20

# ignore clear
:       .lda 80
:       .trb.dp b3

# no battle animation
:       .lda 12
:       .sta.dp b5

# don't retarget
:       .lda 04
:       .sta.dp ba

# fixed damage
:       .stz.addr 3414

# no mp cost
:       .stz.addr 11a5

# non-elemental
:       .stz.addr 11a1

# target self
:       .lda 03
:       .sta.addr 11a0

:       .rep 20

# spell power 1 (required to do any damage)
# clear 11a7 (miss if status fails, indexed battle messages)
:       .lda 0001
:       .sta.addr 11a6

# no hit rate or special effect
:       .stz.addr 11a8

# no status effects
:       .stz.addr 11aa
:       .stz.addr 11ac

# no damage split, ignore defense, ignore reflect
:       .lda 0260
:       .sta.addr 11a2

# can't dodge, zero mp cost
:       .lda 0020
:       .sta.addr 11a4

# get fixed damage value from b8-b9 (normally targeting bytes)
:       .lda.dp b8
:       .and 3fff
:       .sta.addr 11b0

# bit 4000 means MP damage
:       .lda 4000
:       .bit.dp b8
:       .beq not-concerns-mp
:       .lda 8000
:       .tsb.addr 11a2
.label not-concerns-mp

# bit 8000 means healing
:       .lda 8000
:       .bit.dp b8
:       .beq not-healing
:       .lda 0001
:       .tsb.addr 11a4
.label not-healing

# target only self
:       .lda.addr.x 3018
:       .sta.dp b8

:       .jml $reentry-address

$pointer-address
:       $main,2
